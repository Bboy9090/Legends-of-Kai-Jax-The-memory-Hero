// kinetic_engine.ts
export class KineticEngine {
  gravity = 9.8;
  inputBuffer: string[] = [];
  coyoteFrames = 4;
  hitStopFrames = 0;

  update() {
    if (this.hitStopFrames > 0) {
      this.hitStopFrames--;
      return;
    }
  }

  triggerHitStop(frames: number) {
    this.hitStopFrames = frames;
  }

  bufferInput(input: string) {
    if (this.inputBuffer.length > 8) this.inputBuffer.shift();
    this.inputBuffer.push(input);
  }
}
