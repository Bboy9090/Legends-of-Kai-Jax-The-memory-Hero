1) Math Utilities (Vec2/Vec3 + helpers)

// math.ts
export class Vec2 {
  constructor(public x = 0, public y = 0) {}
  clone() { return new Vec2(this.x, this.y); }
  set(x:number,y:number){ this.x=x; this.y=y; return this; }
  add(v:Vec2){ this.x+=v.x; this.y+=v.y; return this; }
  sub(v:Vec2){ this.x-=v.x; this.y-=v.y; return this; }
  mul(s:number){ this.x*=s; this.y*=s; return this; }
  len(){ return Math.hypot(this.x,this.y); }
  norm(){
    const l=this.len() || 1;
    this.x/=l; this.y/=l;
    return this;
  }
  clampLen(max:number){
    const l=this.len();
    if(l>max){ this.norm().mul(max); }
    return this;
  }
}

export class Vec3 {
  constructor(public x = 0, public y = 0, public z = 0) {}
  clone(){ return new Vec3(this.x,this.y,this.z); }
  set(x:number,y:number,z:number){ this.x=x; this.y=y; this.z=z; return this; }
  add(v:Vec3){ this.x+=v.x; this.y+=v.y; this.z+=v.z; return this; }
  sub(v:Vec3){ this.x-=v.x; this.y-=v.y; this.z-=v.z; return this; }
  mul(s:number){ this.x*=s; this.y*=s; this.z*=s; return this; }
  len(){ return Math.hypot(this.x,this.y,this.z); }
  norm(){
    const l=this.len() || 1;
    this.x/=l; this.y/=l; this.z/=l;
    return this;
  }
  static lerp(a:Vec3,b:Vec3,t:number){
    return new Vec3(
      a.x+(b.x-a.x)*t,
      a.y+(b.y-a.y)*t,
      a.z+(b.z-a.z)*t
    );
  }
}

export function clamp(v:number,min:number,max:number){
  return Math.max(min, Math.min(max, v));
}

export function smoothDamp(current:number, target:number, smoothing:number, dt:number){
  // smoothing: bigger = snappier
  const t = 1 - Math.exp(-smoothing * dt);
  return current + (target - current) * t;
}

export function smoothDampVec3(current:Vec3, target:Vec3, smoothing:number, dt:number){
  const t = 1 - Math.exp(-smoothing * dt);
  return Vec3.lerp(current, target, t);
}

2) Mobile Virtual Joystick (DOM, touch math, deadzone, normalized output)

// mobile_joystick.ts
import { Vec2, clamp } from "./math";

export type JoystickOutput = {
  raw: Vec2;        // pixels from center
  value: Vec2;      // normalized [-1..1]
  magnitude: number;// 0..1
  active: boolean;
};

export class MobileJoystick {
  center = new Vec2();
  pointer = new Vec2();
  radius = 70;          // px
  deadzone = 0.12;      // normalized
  active = false;

  private touchId: number | null = null;

  output: JoystickOutput = {
    raw: new Vec2(),
    value: new Vec2(),
    magnitude: 0,
    active: false,
  };

  constructor(
    private baseEl: HTMLElement,
    private knobEl: HTMLElement
  ){
    this.bind();
  }

  private bind(){
    const onStart = (e: TouchEvent) => {
      // take first touch on left side/base element
      const t = e.changedTouches[0];
      this.touchId = t.identifier;
      this.active = true;

      const rect = this.baseEl.getBoundingClientRect();
      const cx = rect.left + rect.width * 0.5;
      const cy = rect.top + rect.height * 0.5;

      this.center.set(cx, cy);
      this.pointer.set(t.clientX, t.clientY);
      this.updateOutput();
    };

    const onMove = (e: TouchEvent) => {
      if(!this.active || this.touchId === null) return;
      for(const t of Array.from(e.changedTouches)){
        if(t.identifier !== this.touchId) continue;
        this.pointer.set(t.clientX, t.clientY);
        this.updateOutput();
      }
    };

    const onEnd = (e: TouchEvent) => {
      if(this.touchId === null) return;
      for(const t of Array.from(e.changedTouches)){
        if(t.identifier !== this.touchId) continue;
        this.touchId = null;
        this.active = false;
        this.output.raw.set(0,0);
        this.output.value.set(0,0);
        this.output.magnitude = 0;
        this.output.active = false;
        this.renderKnob();
      }
    };

    this.baseEl.addEventListener("touchstart", onStart, { passive: false });
    window.addEventListener("touchmove", onMove, { passive: false });
    window.addEventListener("touchend", onEnd, { passive: true });
    window.addEventListener("touchcancel", onEnd, { passive: true });
  }

  private updateOutput(){
    const dx = this.pointer.x - this.center.x;
    const dy = this.pointer.y - this.center.y;

    const raw = new Vec2(dx, dy);
    raw.clampLen(this.radius);

    const nx = raw.x / this.radius;
    const ny = raw.y / this.radius;

    const mag = clamp(Math.hypot(nx, ny), 0, 1);

    // deadzone
    let vx = nx, vy = ny;
    if(mag < this.deadzone){
      vx = 0; vy = 0;
    }

    this.output.raw = raw;
    this.output.value.set(vx, vy);
    this.output.magnitude = mag;
    this.output.active = true;

    this.renderKnob();
  }

  private renderKnob(){
    // Move knob relative to base center
    // knobEl should be position:absolute inside baseEl
    const x = this.output.raw.x;
    const y = this.output.raw.y;
    this.knobEl.style.transform = `translate(${x}px, ${y}px)`;
  }
}

DOM expectation: a joystick base div with a knob div inside. The base can be fixed bottom-left.

4) Character Controller (3D movement + facing + dodge + stamina)

This assumes a simple world where Y is up and the ground plane is XZ.

// mobile_character_controller.ts
import { Vec2, Vec3, clamp, smoothDamp, smoothDampVec3 } from "./math";

export type ControllerInput = {
  move: Vec2;           // from joystick normalized
  attack: boolean;
  heavy: boolean;
  dodge: boolean;
  skill: boolean;
};

export class MobileCharacterController {
  pos = new Vec3(0,0,0);
  vel = new Vec3(0,0,0);
  forward = new Vec3(0,0,1);

  // tuning
  maxSpeed = 6.2;
  accel = 22;
  decel = 26;
  turnSpeed = 14;

  // dodge
  dodging = false;
  dodgeTime = 0;
  dodgeDuration = 0.28;
  dodgeSpeed = 10;

  // stamina
  stamina = 100;
  staminaRegen = 18;
  dodgeCost = 20;

  // combat state light
  attackLock = 0; // seconds remaining for attack animation lock

  update(dt:number, input: ControllerInput){
    // regen stamina
    this.stamina = clamp(this.stamina + this.staminaRegen * dt, 0, 100);

    // tick locks
    this.attackLock = Math.max(0, this.attackLock - dt);

    // dodge trigger
    if(!this.dodging && input.dodge && this.stamina >= this.dodgeCost){
      this.stamina -= this.dodgeCost;
      this.dodging = true;
      this.dodgeTime = 0;
    }

    // movement vector from joystick (camera-relative handled elsewhere)
    const move2 = input.move.clone();
    const moveMag = move2.len();
    if(moveMag > 1) move2.norm();

    // convert 2D input -> 3D direction (XZ)
    // NOTE: you should pass camera-relative move in already (see camera section)
    const desiredDir = new Vec3(move2.x, 0, move2.y);
    if(desiredDir.len() > 0.001) desiredDir.norm();

    // face movement direction (unless locked)
    if(desiredDir.len() > 0.001 && this.attackLock <= 0){
      this.forward = smoothDampVec3(this.forward, desiredDir, this.turnSpeed, dt).norm();
    }

    // dodging overrides standard move
    if(this.dodging){
      this.dodgeTime += dt;
      const t = this.dodgeTime / this.dodgeDuration;

      // dodge direction: if stick held, use it, else forward
      const dodgeDir = (desiredDir.len() > 0.001 ? desiredDir : this.forward).clone();
      this.vel.x = dodgeDir.x * this.dodgeSpeed;
      this.vel.z = dodgeDir.z * this.dodgeSpeed;

      if(t >= 1){
        this.dodging = false;
        // small brake after dodge
        this.vel.x *= 0.35;
        this.vel.z *= 0.35;
      }
    } else {
      // standard accel/decel
      const targetSpeed = this.attackLock > 0 ? 0 : this.maxSpeed * clamp(moveMag, 0, 1);
      const currentSpeed = Math.hypot(this.vel.x, this.vel.z);

      // acceleration factor depends on whether moving or stopping
      const rate = targetSpeed > currentSpeed ? this.accel : this.decel;
      const newSpeed = smoothDamp(currentSpeed, targetSpeed, rate, dt);

      // apply in desired direction (or brake if no input)
      if(desiredDir.len() > 0.001){
        this.vel.x = desiredDir.x * newSpeed;
        this.vel.z = desiredDir.z * newSpeed;
      } else {
        // brake toward 0 while preserving direction
        const dir = new Vec3(this.vel.x, 0, this.vel.z);
        if(dir.len() > 0.001) dir.norm();
        this.vel.x = dir.x * newSpeed;
        this.vel.z = dir.z * newSpeed;
      }
    }

    // integrate
    this.pos.x += this.vel.x * dt;
    this.pos.z += this.vel.z * dt;

    // simple ground lock
    this.pos.y = 0;
  }

  // call when triggering attacks to stop skating during animations
  lockForAttack(seconds:number){
    this.attackLock = Math.max(this.attackLock, seconds);
  }
}

5) Mobile Camera Assist (3rd person, target-aware, zoom-out on danger)

Camera must babysit on mobile.

// mobile_camera_assist.ts
import { Vec3, clamp, smoothDamp, smoothDampVec3 } from "./math";

export type Enemy = { pos: Vec3; alive: boolean };

export class MobileCameraAssist {
  // camera state (virtual)
  camPos = new Vec3(0, 7, -8);
  camLook = new Vec3(0, 0, 0);

  // tuning
  height = 7;
  distance = 8;
  followSmoothing = 10;
  lookSmoothing = 12;

  // zoom out
  baseDistance = 8;
  dangerDistance = 11;

  update(dt:number, playerPos:Vec3, playerForward:Vec3, enemies:Enemy[]){
    const alive = enemies.filter(e=>e.alive);
    const nearest = alive
      .map(e=>({e, d: e.pos.clone().sub(playerPos).len()}))
      .sort((a,b)=>a.d-b.d)[0]?.e ?? null;

    // danger if multiple enemies close
    const closeCount = alive.filter(e=> e.pos.clone().sub(playerPos).len() < 8).length;
    const danger = clamp(closeCount / 3, 0, 1);

    const targetDistance = this.baseDistance + (this.dangerDistance - this.baseDistance) * danger;

    // camera offset behind player
    const back = playerForward.clone().mul(-1);
    const desiredCam = playerPos.clone()
      .add(new Vec3(back.x * targetDistance, this.height, back.z * targetDistance));

    this.camPos = smoothDampVec3(this.camPos, desiredCam, this.followSmoothing, dt);

    // look target: between player and nearest enemy (soft target)
    const look = nearest
      ? playerPos.clone().add(nearest.pos.clone().sub(playerPos).mul(0.35))
      : playerPos.clone();

    this.camLook = smoothDampVec3(this.camLook, look, this.lookSmoothing, dt);
  }
}

6) Camera-Relative Joystick Conversion (this is what makes it feel right)

Joystick gives X/Y on screen. Convert to world movement based on camera forward/right.

// camera_relative.ts
import { Vec2, Vec3 } from "./math";

// cameraForward should be flattened (y=0) and normalized
export function joystickToWorld(move: Vec2, cameraForward: Vec3){
  const f = cameraForward.clone(); f.y = 0; f.norm();
  const r = new Vec3(f.z, 0, -f.x); // right vector from forward
  const out = new Vec3(
    r.x * move.x + f.x * move.y,
    0,
    r.z * move.x + f.z * move.y
  );
  if(out.len() > 0.001) out.norm();
  return out;
}

You then feed that into the controller as input.move = new Vec2(world.x, world.z) or directly use Vec3 movement, whichever you prefer.

1️⃣ Soft Target Lock (Mobile Friendly)

No hard lock-on button. We auto-select best target when attacking.

// soft_target.ts
import { Vec3 } from "./math";

export type Enemy = {
  pos: Vec3;
  alive: boolean;
  hp: number;
};

export function findSoftTarget(
  playerPos: Vec3,
  playerForward: Vec3,
  enemies: Enemy[],
  range = 5
): Enemy | null {

  let best: Enemy | null = null;
  let bestScore = Infinity;

  for(const e of enemies){
    if(!e.alive) continue;

    const to = e.pos.clone().sub(playerPos);
    const dist = to.len();
    if(dist > range) continue;

    to.norm();

    // forward bias
    const dot = playerForward.x * to.x + playerForward.z * to.z;
    const anglePenalty = 1 - dot; // smaller is better

    const score = dist + anglePenalty * 2.5;

    if(score < bestScore){
      bestScore = score;
      best = e;
    }
  }

  return best;
}

This prefers:
	•	Enemies in front
	•	Enemies close
	•	Ignores offscreen junk

Clean.

// attack_magnet.ts
import { Vec3 } from "./math";

export function applyAttackMagnet(
  playerPos: Vec3,
  enemyPos: Vec3,
  maxPull = 1.2
){
  const dir = enemyPos.clone().sub(playerPos);
  const dist = dir.len();
  if(dist < 0.001) return new Vec3();

  dir.norm();

  const pullStrength = Math.min(maxPull, dist * 0.4);

  return dir.mul(pullStrength);
}

Call this on attack startup.

This makes attacks “snap” slightly toward enemies.

Feels intentional instead of clumsy.

export function applyHitReaction(

  damage: number

// combat/MovePlayer.ts
import * as THREE from 'three'
import { MoveSpec } from '../types/MoveSpec'
import { Hurtbox } from './Hurtbox'

export class MovePlayer {
  currentMove: MoveSpec | null = null
  frame: number = 0
  hitboxes: THREE.Mesh[] = []
  scene: THREE.Scene
  hurtbox: Hurtbox
  hitstopFrames: number = 0

  constructor(scene: THREE.Scene, hurtbox: Hurtbox) {
    this.scene = scene
    this.hurtbox = hurtbox
  }

  startMove(move: MoveSpec) {
    this.currentMove = move
    this.frame = 0
  }

  update() {
    if (!this.currentMove) return

    if (this.hitstopFrames > 0) {
      this.hitstopFrames--
      return
    }

    this.clearHitboxes()

    const move = this.currentMove
    this.frame++

    move.hits.forEach(hit => {
      if (this.frame >= hit.startF && this.frame <= hit.endF) {
        const geo = new THREE.BoxGeometry(hit.halfW * 2, hit.halfH * 2, 1)
        const mat = new THREE.MeshBasicMaterial({ wireframe: true, color: 0xff0000 })
        const mesh = new THREE.Mesh(geo, mat)

        mesh.position.set(hit.offX, hit.offY, 0)
        this.scene.add(mesh)
        this.hitboxes.push(mesh)

        this.checkCollision(mesh, hit)
      }
    })

    if (this.frame > move.startup + move.active + move.recovery) {
      this.currentMove = null
    }
  }

  checkCollision(hitbox: THREE.Mesh, hit) {
    const hb = this.hurtbox.mesh

    const hitBB = new THREE.Box3().setFromObject(hitbox)
    const hurtBB = new THREE.Box3().setFromObject(hb)

    if (hitBB.intersectsBox(hurtBB)) {
      this.hurtbox.takeDamage(hit.dmg)

      this.hitstopFrames = this.currentMove?.hitstopOnHit ?? 0

      // simple knockback
      hb.position.x += hit.kbX
      hb.position.y += hit.kbY
    }
  }

  clearHitboxes() {
    this.hitboxes.forEach(h => this.scene.remove(h))
    this.hitboxes = []
  }
}

export type MissionState = "idle" | "running" | "won" | "lost"

export class MissionTracker {
  state: MissionState = "idle"
  kills = 0
  bossDefeated = false
  onWin?: () => void
  onLose?: () => void

  start() { this.state = "running" }

  registerKill() { this.kills++ }

  registerBossDefeated() {
    this.bossDefeated = true
    this.state = "won"
    this.onWin?.()
  }

  registerPlayerDead() {
    this.state = "lost"
    this.onLose?.()

// mission/WaveDirector.ts
import { MissionTracker } from "./MissionTracker"

type EnemyType = "fang_grunt" | "covenant_scout" | "covenant_enforcer"

export class WaveDirector {
  private scene: any
  private tracker: MissionTracker
  private activeEnemies: any[] = []

  constructor(scene: any, tracker: MissionTracker) {
    this.scene = scene
    this.tracker = tracker
  }

  spawnEnemy(type: EnemyType, x: number, y: number) {
    const enemy = { type, hp: 30, x, y, isBoss: type === "covenant_enforcer" }
    this.activeEnemies.push(enemy)
    return enemy
  }

  update() {
    // remove dead enemies
    const before = this.activeEnemies.length
    this.activeEnemies = this.activeEnemies.filter(e => e.hp > 0)
    const after = this.activeEnemies.length

    if (after < before) this.tracker.registerKill()
  }

  markEnemyDead(enemy: any) {
    enemy.hp = 0
    if (enemy.isBoss) this.tracker.registerBossDefeated()
  }
}

🥊 3️⃣ Move Interpreter (The Core)

This is the subsystem you’ve been missing.

// combat/MovePlayer.ts
import * as THREE from 'three'
import { MoveSpec } from '../types/MoveSpec'
import { Hurtbox } from './Hurtbox'

export class MovePlayer {
  currentMove: MoveSpec | null = null
  frame: number = 0
  hitboxes: THREE.Mesh[] = []
  scene: THREE.Scene
  hurtbox: Hurtbox
  hitstopFrames: number = 0

  constructor(scene: THREE.Scene, hurtbox: Hurtbox) {
    this.scene = scene
    this.hurtbox = hurtbox
  }

  startMove(move: MoveSpec) {
    this.currentMove = move
    this.frame = 0
  }

  update() {
    if (!this.currentMove) return

    if (this.hitstopFrames > 0) {
      this.hitstopFrames--
      return
    }

    this.clearHitboxes()

    const move = this.currentMove
    this.frame++

    move.hits.forEach(hit => {
      if (this.frame >= hit.startF && this.frame <= hit.endF) {
        const geo = new THREE.BoxGeometry(hit.halfW * 2, hit.halfH * 2, 1)
        const mat = new THREE.MeshBasicMaterial({ wireframe: true, color: 0xff0000 })
        const mesh = new THREE.Mesh(geo, mat)

        mesh.position.set(hit.offX, hit.offY, 0)
        this.scene.add(mesh)
        this.hitboxes.push(mesh)

        this.checkCollision(mesh, hit)
      }
    })

    if (this.frame > move.startup + move.active + move.recovery) {
      this.currentMove = null
    }
  }

  checkCollision(hitbox: THREE.Mesh, hit) {
    const hb = this.hurtbox.mesh

    const hitBB = new THREE.Box3().setFromObject(hitbox)
    const hurtBB = new THREE.Box3().setFromObject(hb)

    if (hitBB.intersectsBox(hurtBB)) {
      this.hurtbox.takeDamage(hit.dmg)

      this.hitstopFrames = this.currentMove?.hitstopOnHit ?? 0

      // simple knockback
      hb.position.x += hit.kbX
      hb.position.y += hit.kbY
    }
  }

  clearHitboxes() {
    this.hitboxes.forEach(h => this.scene.remove(h))
    this.hitboxes = []
  }
}
